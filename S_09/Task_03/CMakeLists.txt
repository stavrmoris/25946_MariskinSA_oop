cmake_minimum_required(VERSION 3.14)
project(oop09_homework LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/utf-8 /W4)
else()
    add_compile_options(-Wall -Wextra -pedantic)
endif()
add_library(robot_support STATIC robot_access.cpp robot_squad.cpp)
target_include_directories(robot_support PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(robot_app main.cpp)
target_link_libraries(robot_app PRIVATE robot_support)
