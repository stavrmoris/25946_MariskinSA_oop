cmake_minimum_required(VERSION 3.14)
project(oop12_classwork LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/utf-8 /W4)
else()
    add_compile_options(-Wall -Wextra -pedantic)
endif()
add_library(human STATIC human.cpp student.cpp)
target_include_directories(human PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(human_app main.cpp)
target_link_libraries(human_app PRIVATE human)
